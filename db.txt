CREATE TABLE HORARIOS (
    id_horario INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    id_empresa INT NOT NULL,
    fecha DATE NOT NULL,
    tipo_jornada ENUM('TRABAJO', 'VACACIONES', 'MEDICO', 'LIBRE', 'FESTIVO') DEFAULT 'TRABAJO',
    hora_inicio TIME NULL,
    hora_fin TIME NULL,
    horas_totales DECIMAL(4,2) NULL,
    observaciones TEXT NULL,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (id_usuario) REFERENCES USUARIO(id_usuario) ON DELETE CASCADE,
    FOREIGN KEY (id_empresa) REFERENCES EMPRESA(id_empresa) ON DELETE CASCADE,
    UNIQUE KEY unico_fecha_usuario (id_usuario, id_empresa, fecha),
    INDEX idx_fecha (fecha),
    INDEX idx_usuario_fecha (id_usuario, fecha)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='Horarios aprobados por fecha';

-- TABLA: SOLICITUDES_HORARIO
-- Almacena las peticiones pendientes de validación
CREATE TABLE SOLICITUDES_HORARIO (
    id_solicitud INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    id_empresa INT NOT NULL,
    tipo_solicitud ENUM('HORARIO_MES', 'VACACIONES', 'MEDICO') NOT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_fin DATE NULL,
    estado ENUM('PENDIENTE', 'APROBADO', 'RECHAZADO') DEFAULT 'PENDIENTE',
    motivo_rechazo TEXT NULL,
    observaciones TEXT NULL,
    validado_por INT NULL,
    validado_en TIMESTAMP NULL,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_usuario) REFERENCES USUARIO(id_usuario) ON DELETE CASCADE,
    FOREIGN KEY (id_empresa) REFERENCES EMPRESA(id_empresa) ON DELETE CASCADE,
    FOREIGN KEY (validado_por) REFERENCES USUARIO(id_usuario) ON DELETE SET NULL,
    INDEX idx_estado (estado),
    INDEX idx_usuario (id_usuario),
    INDEX idx_empresa (id_empresa),
    INDEX idx_fecha_inicio (fecha_inicio)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='Solicitudes de horarios pendientes de validación';

-- TABLA: DETALLE_SOLICITUD_HORARIO
-- Detalle día a día de cada solicitud de horario mensual
CREATE TABLE DETALLE_SOLICITUD_HORARIO (
    id_detalle INT AUTO_INCREMENT PRIMARY KEY,
    id_solicitud INT NOT NULL,
    fecha DATE NOT NULL,
    tipo_jornada ENUM('TRABAJO', 'VACACIONES', 'MEDICO', 'LIBRE', 'FESTIVO') DEFAULT 'TRABAJO',
    hora_inicio TIME NULL,
    hora_fin TIME NULL,
    horas_totales DECIMAL(4,2) NULL,
    observaciones TEXT NULL,
    FOREIGN KEY (id_solicitud) REFERENCES SOLICITUDES_HORARIO(id_solicitud) ON DELETE CASCADE,
    INDEX idx_solicitud (id_solicitud),
    INDEX idx_fecha (fecha)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='Detalle diario de solicitudes de horario';

-- TABLA: HISTORIAL_VALIDACIONES
-- Registro de todas las validaciones realizadas (auditoría)
CREATE TABLE HISTORIAL_VALIDACIONES (
    id_historial INT AUTO_INCREMENT PRIMARY KEY,
    id_solicitud INT NOT NULL,
    accion ENUM('APROBADO', 'RECHAZADO', 'MODIFICADO') NOT NULL,
    motivo TEXT NULL,
    validado_por INT NOT NULL,
    validado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_solicitud) REFERENCES SOLICITUDES_HORARIO(id_solicitud) ON DELETE CASCADE,
    FOREIGN KEY (validado_por) REFERENCES USUARIO(id_usuario) ON DELETE CASCADE,
    INDEX idx_solicitud (id_solicitud),
    INDEX idx_validado_por (validado_por)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='Historial de validaciones de horarios';

-- Tabla para guardar los reportes generados
CREATE TABLE REPORTES (
    id_reporte INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    id_empresa INT NOT NULL,
    tipo_reporte VARCHAR(50) NOT NULL,
    mes INT NOT NULL,
    anio INT NOT NULL,
    fecha_generacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    generado_por INT NOT NULL,
    ruta_archivo VARCHAR(255) NULL,
    FOREIGN KEY (id_usuario) REFERENCES USUARIO(id_usuario) ON DELETE CASCADE,
    FOREIGN KEY (id_empresa) REFERENCES EMPRESA(id_empresa) ON DELETE CASCADE,
    FOREIGN KEY (generado_por) REFERENCES USUARIO(id_usuario) ON DELETE CASCADE,
    INDEX idx_usuario_fecha (id_usuario, anio, mes),
    INDEX idx_empresa (id_empresa)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='Historial de reportes generados';
